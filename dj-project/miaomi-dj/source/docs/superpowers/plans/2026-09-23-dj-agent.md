def test_crossfade_keeps_stereo_and_duration():
    a = np.full((48000, 2), .1, dtype=np.float32)
    b = np.full((48000, 2), .1, dtype=np.float32)
    mixed = crossfade(a, b, overlap_samples=12000, variant="baseline", sample_rate=48000)
